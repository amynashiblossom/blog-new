# A. 젬스 지침 (여기부터 A장 끝까지 복사)

너는 샵솔 스팟워크 공고에 지원한 사람들을 채점하는 계산기다. **누가 좋은 사람인지 판단하지 않는다.** 아래 배점표와 결정 규칙을 그대로 실행만 한다.

## 금지
- 규칙에 없는 가점·감점, 빈 값 추정(계산할 수 없을 때만 정해진 중립값), 반올림, 외부 정보 조회, 입력에 없는 정보 사용
- 성별·나이·사진은 입력에 없으며 어떤 추론에도 쓰지 않는다

## 입력
공고 1건과 지원자 스냅샷 목록(JSON). 모든 값은 지원 시각 기준으로 고정돼 있다. 결과 값: 완료 / 무단결근 / 취소 / 사업주취소.

## 절차 (지원자마다 순서대로)
1. **기준 시각**: 지원시각. "최근 N개월" = 지원시각에서 달력 N개월 전 같은 시각부터(포함) 지원시각 직전까지. 날짜가 없으면 그 달 말일
2. **취소 판정**: 결과가 "취소"인 건마다 — 사유가 사망(부고장, 근무일 +7일 이내)이나 진료(진료확인서, 근무일 +2일 이내)이고, 증빙확인시각이 기한 안이면서 지원시각 이전이고, 같은 달력 월(근무 시작일 기준)에 먼저 인정된 취소가 없으면 **인정 취소**. 나머지 취소는 전부 **확정 후 취소**. 시간순으로 판정한다
3. **0번 규칙**: 결격 이력 = 스팟 무단결근 + 확정 후 취소 + 정규무단결근. **기간과 상관없이 1건이라도 있으면 그룹 C.** 없으면, 공고 시간(시작~종료)과 샵솔근무·확정스팟이 1분이라도 겹치면(앞 근무 종료 = 공고 시작은 겹침 아님) 그룹 B. 아니면 그룹 A
4. **항목 채점** (아래 배점표). 2순위 긴급·직전을 셀 때는 먼저 최근 3개월 **완료** 건을 한 줄씩 적고, 긴급은 등록시각 · 직전은 지원시각 칸이 **비어 있으면 그 건 옆에 "시각 없음 → 제외"라고 적은 뒤** 남은 건만 센다 (빈 칸을 다른 시각으로 채우지 않는다)
5. **합산**: 소계 4개, 총점. 반올림 없이 소수 첫째 자리까지 정확히
6. **정렬**: 그룹 A → B → C. 그룹 안은 총점 높은 순 → 0순위 소계 → 1순위 소계 → 지원시각 빠른 순 → 지원ID 오름차순
7. **추천이유 한 줄**

# D. 채점 절차서 · 입력 필드 정의서

## 1. 한 줄 요약

AI는 **누가 좋은 사람인지 판단하지 않는다.** 기획자가 사장님 UT로 정한 판단 순서를 **7단계 절차**대로 실행만 한다. 입력은 지원 시각에 고정한 스냅샷 하나이고, 절차의 모든 갈림길에는 결정 규칙이 붙어 있어서 같은 입력이면 누가(사람이든 AI든) 언제 채점해도 같은 결과가 나온다.

---

## 2. 입력 필드 정의서 — AI가 받는 것 (지원자 1명 = 스냅샷 1개)

모든 값은 **지원 시각**에 서버가 모아 고정한다(규칙 1·8). AI는 외부를 다시 조회하지 않는다. 성별·나이·사진은 입력에 없다.

# E. 기준 채점기 코드 (참고용, Python)

```python
"""위솝xTeamK AI 추천채용 기준 채점기 (결정 규칙 최종 9/25 구현, 기대값 산출용)"""
import json, math, calendar
from datetime import datetime, timedelta, date

def dt(s): return datetime.strptime(s, "%Y-%m-%d %H:%M")
def d(s): return datetime.strptime(s, "%Y-%m-%d").date()

def minus_months(t, n):
    y, m = t.year, t.month - n
    while m <= 0: m += 12; y -= 1
    day = min(t.day, calendar.monthrange(y, m)[1])
    return t.replace(year=y, month=m, day=day)

def full_months(a, b):  # a=가입일(date), b=지원시각(datetime)
    m = (b.year - a.year) * 12 + (b.month - a.month)
    if b.day < a.day: m -= 1
    return max(m, 0)

def within(ts, apply, n):  # 지원 시각 기준 최근 n개월 (시작 포함, 지원 직전까지)
    return minus_months(apply, n) <= ts < apply

def band_int(v, table):
    for lo, hi, sc in table:
        if (lo is None or v >= lo) and (hi is None or v <= hi): return sc
    raise ValueError(v)

def score(job, a):
    apply = dt(a["지원시각"]); start = dt(job["시작"]); end = dt(job["종료"])
    out = {"지원자": a["이름"], "지원ID": a["지원ID"]}
    spots = [dict(e, 시작=dt(e["시작"])) for e in a.get("스팟이력", [])]
    for e in spots:
        if e.get("등록시각"): e["등록시각"] = dt(e["등록시각"])
        if e.get("지원시각"): e["지원시각"] = dt(e["지원시각"])
    regs = a.get("정규이력", [])
    reg_abs = [dt(x) for x in a.get("정규무단결근", [])]

    # --- 인정 취소 판정 (규칙 7: 월 1회, 기한 내 증빙, 지원 시각에 확인된 것만) ---
    used = set()
    for e in sorted(spots, key=lambda e: e["시작"]):
        if e["결과"] == "취소":
            ok = False
            if e.get("사유") in ("사망", "진료") and e.get("증빙확인시각"):
                lim = e["시작"].date() + timedelta(days=7 if e["사유"] == "사망" else 2)
                chk = dt(e["증빙확인시각"])
                key = (e["시작"].year, e["시작"].month)
                if chk.date() <= lim and chk <= apply and key not in used:
                    ok = True; used.add(key)
            e["결과"] = "인정취소" if ok else "확정후취소"

    # --- 0번 규칙 ① 결격 (최근 12개월, 스팟 무단결근·확정 후 취소 + 정규 무단결근) ---
    nd12 = sum(1 for e in spots if e["결과"] == "무단결근" and e["시작"] < apply) + \
           sum(1 for t in reg_abs if t < apply)   # 9/25 16:25: 기간 무관(전체 기간), 기록은 사라지지 않음
    cx12 = sum(1 for e in spots if e["결과"] == "확정후취소" and e["시작"] < apply)
    disq = nd12 + cx12 > 0
    # --- 0번 규칙 ② 겹침 (1분이라도, 끝=시작은 아님) ---
    overlap = any(dt(s["시작"]) < end and dt(s["종료"]) > start for s in a.get("샵솔근무", []) + a.get("확정스팟", []))
    group = "C" if disq else ("B" if overlap else "A")

    # ===== 0순위 =====
    comp = sum(1 for e in spots if e["결과"] == "완료")
    den = sum(1 for e in spots if e["결과"] in ("완료", "무단결근", "확정후취소"))
    if den < 3: s1 = 9.0; r1 = None
    else:
        r1 = math.floor(comp * 100 / den)
        s1 = band_int(r1, [(100, None, 18.0), (95, 99, 15.0), (90, 94, 12.0), (80, 89, 8.0), (70, 79, 4.0), (None, 69, 0.0)])
    s2 = 0.0 if reg_abs else 6.0
    mon = None  # 9/26 밤: 가입기간 미사용
    s3 = 0.0  # 9/26 밤: 샵솔 가입기간 삭제 (4점은 #3으로 이동)
    s4 = band_int(comp, [(20, None, 8.0), (10, 19, 6.0), (5, 9, 4.0), (1, 4, 2.0), (0, 0, 0.0)])  # 9/26: 0-3 = 스팟 완료 건수(전체 기간)
    wd = [(d(x["날짜"]), x["결과"]) for x in a.get("최근근무일", []) if minus_months(apply, 3).date() <= d(x["날짜"]) < apply.date() or (d(x["날짜"]) == apply.date() and False)]
    if not wd: s5 = 1.5; streak = None
    else:
        days = {}
        for day, res in wd: days[day] = days.get(day, "정상") if res == "정상" else "결격"
        streak = 0
        for day in sorted(days, reverse=True):
            if days[day] == "결격": break
            streak += 1
        s5 = band_int(streak, [(30, None, 3.0), (15, 29, 2.0), (5, 14, 1.0), (0, 4, 0.0)])
    same_store_done = sum(1 for e in spots if e["결과"] == "완료" and e["매장ID"] == job["매장ID"] and within(e["시작"], apply, 24)) + \
                      sum(r.get("24개월완료횟수", 0) for r in regs if r["매장ID"] == job["매장ID"])
    s6 = 3.0 if same_store_done >= 1 else 0.0  # 9/26: 이 매장 근무 경험(1회+)
    p = a.get("백분위")
    if not p or not p.get("모집단포함"): s7 = 1.0; pct = None

        s7 = band_int(pct, [(0, 10, 2.0), (11, 30, 1.5), (31, 50, 1.0), (51, None, 0.5)])
    p0 = [s1, s2, s3, s4, s5, s6, s7]

    # ===== 1순위 동심원 (24개월 완료, 단계마다 정규 우선, ID 없으면 건너뜀) =====
    def level(w):
        if w["매장ID"] == job["매장ID"]: return 0
        if w.get("브랜드ID") and job.get("브랜드ID") and w["브랜드ID"] == job["브랜드ID"]: return 1
        if w.get("계열ID") and job.get("계열ID") and w["계열ID"] == job["계열ID"]: return 2
        if w.get("업종코드") == job["업종코드"]: return 3
        return None
    works = [(level(e), "스팟") for e in spots if e["결과"] == "완료" and within(e["시작"], apply, 24)]
    works += [(level(r), "정규") for r in regs if r.get("재직중") or r.get("24개월완료횟수", 0) > 0]
    pts = {0: (11.0, 9.8), 1: (4.5, 4.4), 2: (3.5, 3.4), 3: (2.5, 2.4)}

    kw = 1.0 if a.get("키워드수_24개월", 0) >= 1 else 0.0

    base = dt(a["첫알림시각"]) if a.get("첫알림시각") and dt(a["첫알림시각"]) <= apply else dt(job["공고등록시각"])
    rmin = (apply - base).total_seconds() / 60

    if a.get("매칭활용동의"):
        cands = [s for s in a.get("샵솔근무", []) if start - timedelta(hours=3) <= dt(s["종료"]) <= start]
        if cands:
            s = max(cands, key=lambda s: dt(s["종료"]))
            origin = ("샵솔근무", dt(s["종료"]), s["이동분"])
        elif a.get("GPS") and a["GPS"].get("켜짐"):
            origin = ("GPS", max(apply, start - timedelta(hours=1)), a["GPS"]["이동분"])
    if origin is None and a.get("등록주소"):
        origin = ("등록주소", max(apply, start - timedelta(hours=1)), a["등록주소"]["이동분"])
    if origin is None: sa = 4.0; margin = None

        margin = (start - (origin[1] + timedelta(minutes=origin[2]))).total_seconds() / 60
        sa = 8.0 if margin >= 30 else 6.0 if margin >= 10 else 3.0 if margin >= 0 else 0.0
    done3 = [e for e in spots if e["결과"] == "완료" and within(e["시작"], apply, 3)]

    tail = f" · 추천서 {nrec}건" if nrec else ""
    if disq:

    elif overlap: reason = "샵솔 근무시간과 겹쳐요"
    else:
        inner = min([lv for lv, f in works if lv is not None and not (lv in idkey and not job.get(idkey[lv]))], default=None)
        names = {1: "브랜드", 2: "계열", 3: "업종"}
        if inner == 0:
            if any(lv == 0 and f == "정규" and next((r for r in regs if r["매장ID"] == job["매장ID"] and r.get("재직중")), None) for lv, f in works):
                reason = "우리 매장에서 정규직으로 근무 중이에요" + tail
            else:
                n = sum(1 for e in spots if e["결과"] == "완료" and e["매장ID"] == job["매장ID"] and within(e["시작"], apply, 24))

        elif inner is not None:

            reason = (f"같은 {names[inner]} 매장에서 정규직으로 근무 중이에요" if cur else f"같은 {names[inner]} 매장에서 근무 경험이 있어요") + tail
        elif origin and origin[0] == "샵솔근무" and margin is not None and margin >= 0:

        else:
            reason = (f"무단결근 없이 {comp}회 근무했어요" if comp else "첫 스팟워크 지원이에요") + tail

    out.update({"그룹": group, "총점": tot10 / 10,

def rank(job, apps):
    rs = [score(job, a) for a in apps]
    order = {"A": 0, "B": 1, "C": 2}
    rs.sort(key=lambda r: (order[r["그룹"]], -r["총점"], -r["소계"]["0순위"], -r["소계"]["1순위"], r["지원시각"], r["지원ID"]))
    for i, r in enumerate(rs, 1): r["순위"] = i
    return rs
```
